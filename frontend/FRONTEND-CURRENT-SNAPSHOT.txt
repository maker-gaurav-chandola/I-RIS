import { useEffect, useState, type FormEvent, type ReactElement } from "react";
import {
  Dashboard,
  type DashboardAction,
  type DashboardUser,
} from "./components/Dashboard";
import {
  InterviewConfiguration,
  initialConfiguration,
  type ConfigurationAction,
  type InterviewConfiguration as InterviewConfigurationState,
} from "./components/InterviewConfiguration";
import {
  hasValidConfiguration,
  ReadyScreen,
  type ReadyAction,
} from "./components/ReadyScreen";
import { InterviewSessionEntry } from "./components/InterviewSessionEntry";
import { ActiveInterview } from "./components/ActiveInterview";
import {
  HistoryProgress,
  type HistoryItem,
  type Progress,
} from "./components/HistoryProgress";
import {
  OverviewIcon,
  StartIcon,
  HistoryIcon,
  ProgressIcon,
} from "./components/icons";

const apiBaseUrl = import.meta.env.VITE_API_BASE_URL ?? "http://localhost:3001";

type User = DashboardUser;
type AuthMode = "login" | "register";

export function getPageForPath(pathname: string): string {
  const pages: Record<string, string> = {
    "/": "Entry",
    "/account": "Account Access",
    "/dashboard": "Dashboard",
    "/interview/setup": "Interview Setup",
    "/ready": "Ready",
    "/interview": "Interview",
    "/results": "Results",
    "/history": "History",
    "/progress": "Progress",
  };

  return pages[pathname] ?? "Entry";
}

export function App(): ReactElement {
  const [health, setHealth] = useState("Checking backend...");
  const [user, setUser] = useState<User | null>(null);
  const [authChecked, setAuthChecked] = useState(false);
  const [authMode, setAuthMode] = useState<AuthMode>("login");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [message, setMessage] = useState("");
  const [page, setPage] = useState(getPageForPath(window.location.pathname));
  const [configuration, setConfiguration] =
    useState<InterviewConfigurationState>(initialConfiguration);
  const [sessionId, setSessionId] = useState<string | null>(null);
  const [recentHistory, setRecentHistory] = useState<HistoryItem[]>([]);
  const [recentProgress, setRecentProgress] = useState<Progress | null>(null);

  useEffect(() => {
    fetch(`${apiBaseUrl}/api/health`)
      .then((response) => response.json())
      .then((result: { status?: string; database?: string }) => {
        setHealth(
          result.status === "ok" && result.database === "connected"
            ? "Backend and database connected"
            : "Backend unavailable",
        );
      })
      .catch(() => setHealth("Backend unavailable"));

    fetch(`${apiBaseUrl}/api/auth/me`, { credentials: "include" })
      .then(async (response) =>
        response.ok ? ((await response.json()).user as User) : null,
      )
      .then((authenticatedUser) => {
        setUser(authenticatedUser);
        setAuthChecked(true);
      })
      .catch(() => setAuthChecked(true));
  }, []);

  useEffect(() => {
    if (!user) return;
    fetch(`${apiBaseUrl}/api/interview-history`, { credentials: "include" })
      .then(async (response) =>
        response.ok
          ? ((await response.json()) as {
              history?: HistoryItem[];
              progress?: Progress;
            })
          : null,
      )
      .then((result) => {
        setRecentHistory(result?.history ?? []);
        setRecentProgress(result?.progress ?? null);
      })
      .catch(() => {
        setRecentHistory([]);
        setRecentProgress(null);
      });
  }, [user, page]);

  function navigate(path: string): void {
    window.history.pushState({}, "", path);
    setPage(getPageForPath(path));
    setMessage("");
  }

  async function submitAuth(event: FormEvent<HTMLFormElement>): Promise<void> {
    event.preventDefault();
    setMessage("");
    const response = await fetch(`${apiBaseUrl}/api/auth/${authMode}`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      credentials: "include",
      body: JSON.stringify({ email, password }),
    });
    const result = (await response.json()) as { user?: User; error?: string };
    if (!response.ok || !result.user) {
      setMessage(result.error ?? "Account access could not be completed.");
      return;
    }
    setUser(result.user);
    setPassword("");
    navigate("/dashboard");
  }

  async function logout(): Promise<void> {
    await fetch(`${apiBaseUrl}/api/auth/logout`, {
      method: "POST",
      credentials: "include",
    });
    setUser(null);
    navigate("/account");
  }

  function handleDashboardAction(action: DashboardAction): void {
    if (action === "new-interview") {
      navigate("/interview/setup");
      return;
    }
    if (action === "history") {
      navigate("/history");
      return;
    }
    navigate("/progress");
  }

  function handleConfigurationAction(
    action: ConfigurationAction,
    _configuration?: InterviewConfigurationState,
  ): void {
    if (action === "back") {
      navigate("/dashboard");
      return;
    }
    if (configuration) navigate("/ready");
  }

  async function handleReadyAction(action: ReadyAction): Promise<void> {
    if (action === "edit") navigate("/interview/setup");
    if (action === "start") {
      const response = await fetch(`${apiBaseUrl}/api/interview-sessions`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        credentials: "include",
        body: JSON.stringify({
          ...configuration,
          duration: Number(configuration.duration),
        }),
      });
      const result = (await response.json()) as {
        session?: { id: string };
        error?: string;
      };
      if (!response.ok || !result.session) {
        setMessage(result.error ?? "Interview could not be started.");
        return;
      }
      setSessionId(result.session.id);
      navigate("/interview");
    }
  }

  if (!authChecked) {
    return (
      <main className="shell">
        <p className="status">Checking account access...</p>
      </main>
    );
  }

  if (!user) {
    return (
      <main className="shell">
        <header className="header">
          <div>
            <p className="eyebrow">Interview Response Intelligence System</p>
            <h1>I-RIS</h1>
          </div>
          <span className="status">{health}</span>
        </header>
        <section className="content">
          <p className="eyebrow">
            {page === "Entry" ? "Practice with purpose" : "Account access"}
          </p>
          <h2>
            {page === "Entry"
              ? "Practice Interviews. Improve Every Time."
              : authMode === "login"
                ? "Welcome back"
                : "Create your account"}
          </h2>
          {page === "Entry" ? (
            <>
              <p>
                Build confidence through realistic technical interview practice
                and structured feedback.
              </p>
              <button type="button" onClick={() => navigate("/account")}>
                Continue to account access
              </button>
            </>
          ) : (
            <form className="auth-form" onSubmit={submitAuth}>
              <label>
                Email
                <input
                  type="email"
                  value={email}
                  onChange={(event) => setEmail(event.target.value)}
                  required
                  autoComplete="email"
                />
              </label>
              <label>
                Password
                <input
                  type="password"
                  value={password}
                  onChange={(event) => setPassword(event.target.value)}
                  required
                  autoComplete={
                    authMode === "login" ? "current-password" : "new-password"
                  }
                />
              </label>
              <button type="submit">
                {authMode === "login" ? "Continue" : "Create account"}
              </button>
              {message && (
                <p className="error" role="alert">
                  {message}
                </p>
              )}
              <button
                className="link-button"
                type="button"
                onClick={() => {
                  setAuthMode(authMode === "login" ? "register" : "login");
                  setMessage("");
                }}
              >
                {authMode === "login"
                  ? "Create a new account"
                  : "Use an existing account"}
              </button>
            </form>
          )}
        </section>
      </main>
    );
  }

  return (
    <main className="app-shell">
      <aside className="app-rail" aria-label="Primary navigation">
        <button
          className="rail-brand"
          type="button"
          onClick={() => navigate("/dashboard")}
          aria-label="Go to dashboard"
        >
          <span className="brand-mark">IR</span>
          <span>I-RIS</span>
        </button>
        <p className="rail-label">Workspace</p>
        <nav className="rail-nav">
          {[
            {
              label: "Overview",
              path: "/dashboard",
              icon: <OverviewIcon />,
              active: page === "Dashboard",
            },
            {
              label: "Start interview",
              path: "/interview/setup",
              icon: <StartIcon />,
              active:
                page === "Interview Setup" ||
                page === "Ready" ||
                page === "Interview",
            },
            {
              label: "History",
              path: "/history",
              icon: <HistoryIcon />,
              active: page === "History",
            },
            {
              label: "Progress",
              path: "/progress",
              icon: <ProgressIcon />,
              active: page === "Progress",
            },
          ].map((item) => (
            <a
              key={item.path}
              className={item.active ? "is-active" : ""}
              href={item.path}
              aria-current={item.active ? "page" : undefined}
              onClick={(event) => {
                event.preventDefault();
                navigate(item.path);
              }}
            >
              {item.icon}
              {item.label}
            </a>
          ))}
        </nav>
        <div className="rail-account">
          <span className="account-dot" aria-hidden="true">
            {user.email.slice(0, 1).toUpperCase()}
          </span>
          <span className="account-email">{user.email}</span>
          <button type="button" onClick={logout}>
            Log out
          </button>
        </div>
      </aside>
      <section className="workspace">
        <header className="workspace-header">
          <div>
            <p className="workspace-breadcrumb">I-RIS / {page}</p>
            <p className="workspace-health">
              <span />
              System ready for practice
            </p>
          </div>
          <div className="workspace-context">
            Interview Response Intelligence System
          </div>
        </header>
        <div className="workspace-content">
          {page === "Interview Setup" ? (
            <InterviewConfiguration
              configuration={configuration}
              onChange={setConfiguration}
              onAction={handleConfigurationAction}
            />
          ) : page === "Ready" && hasValidConfiguration(configuration) ? (
            <ReadyScreen
              configuration={configuration}
              onAction={handleReadyAction}
            />
          ) : page === "Interview" && sessionId ? (
            <ActiveInterview apiBaseUrl={apiBaseUrl} sessionId={sessionId} />
          ) : page === "History" || page === "Progress" ? (
            <HistoryProgress
              apiBaseUrl={apiBaseUrl}
              initialView={page === "Progress" ? "progress" : "history"}
              onRePractice={() => navigate("/interview/setup")}
            />
          ) : page === "Interview" ? (
            <InterviewSessionEntry />
          ) : (
            <>
              <Dashboard
                user={user}
                history={recentHistory}
                progress={recentProgress}
                onAction={handleDashboardAction}
              />
              {message && (
                <p className="dashboard-message" role="status">
                  {message}
                </p>
              )}
            </>
          )}
        </div>
      </section>
    </main>
  );
}

import { Navigate, Outlet, useLocation } from "react-router-dom";
import { useAuth } from "@/hooks/useAuth";
import { Spinner } from "@/components/common/Spinner";

// Route protection is a UX convenience only — it is not the security
// boundary. The backend must reject unauthenticated/unauthorized requests
// on every protected endpoint regardless of what this component does.
//
// "checking" is the brief window while GET /api/auth/me is in flight on
// app load (see useAuth) — waiting it out here is what stops a valid
// session from being bounced to /login on every page refresh.
export function ProtectedRoute() {
  const { user, status } = useAuth();
  const location = useLocation();

  if (status === "checking") {
    return <Spinner label="Checking session" />;
  }

  if (!user) {
    return <Navigate to="/login" state={{ from: location }} replace />;
  }

  return <Outlet />;
}

export function ActiveInterview({
  apiBaseUrl,
  sessionId,
}: ActiveInterviewProps): ReactElement {
  const [session, setSession] = useState<Session | null>(null);
  const [answer, setAnswer] = useState("");
  const [closingScore, setClosingScore] = useState("");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(true);
  const [submitting, setSubmitting] = useState(false);
  const [submittingScore, setSubmittingScore] = useState(false);
  const [evaluation, setEvaluation] = useState<EvaluationResult | null>(null);
  const [loadingResults, setLoadingResults] = useState(false);

  async function loadSession(): Promise<void> {
    const response = await fetch(
      `${apiBaseUrl}/api/interview-sessions/${sessionId}`,
      { credentials: "include" },
    );
    const result = (await response.json()) as {
      session?: Session;
      error?: string;
    };
    if (!response.ok || !result.session) {
      setError(result.error ?? "Interview session could not be loaded.");
    } else {
      setSession(result.session);
    }
    setLoading(false);
  }

  useEffect(() => {
    void loadSession();
  }, [sessionId]);

  async function submitAnswer(
    event: FormEvent<HTMLFormElement>,
  ): Promise<void> {
    event.preventDefault();
    if (!answer.trim()) {
      setError("Answer cannot be empty.");
      return;
    }
    setError("");
    setSubmitting(true);
    const response = await fetch(
      `${apiBaseUrl}/api/interview-sessions/${sessionId}/answers`,
      {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        credentials: "include",
        body: JSON.stringify({ answer }),
      },
    );
    const result = (await response.json()) as {
      session?: Session;
      error?: string;
    };
    setSubmitting(false);
    if (!response.ok || !result.session) {
      setError(result.error ?? "Answer could not be submitted.");
      return;
    }
    setSession(result.session);
    setAnswer("");
  }

  async function terminate(): Promise<void> {
    setError("");
    const response = await fetch(
      `${apiBaseUrl}/api/interview-sessions/${sessionId}/close`,
      {
        method: "POST",
        credentials: "include",
      },
    );
    const result = (await response.json()) as {
      session?: Session;
      error?: string;
    };
    if (!response.ok || !result.session) {
      setError(result.error ?? "Interview could not be closed.");
      return;
    }
    setSession(result.session);
  }

  async function submitExpectedScore(): Promise<void> {
    const numericScore = Number(closingScore);
    if (
      !Number.isFinite(numericScore) ||
      numericScore < 0 ||
      numericScore > 100
    ) {
      setError("Provide a valid expected score from 0 to 100.");
      return;
    }
    setError("");
    setSubmittingScore(true);
    const response = await fetch(
      `${apiBaseUrl}/api/interview-sessions/${sessionId}/expected-score`,
      {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        credentials: "include",
        body: JSON.stringify({ expectedScore: numericScore }),
      },
    );
    const result = (await response.json()) as {
      session?: Session;
      error?: string;
    };
    setSubmittingScore(false);
    if (!response.ok || !result.session) {
      setError(result.error ?? "Expected score could not be submitted.");
      return;
    }
    setSession(result.session);
    setClosingScore("");
  }

    setLoadingResults(true);

              max="100"
              step="0.1"

              onChange={(event) => setClosingScore(event.target.value)}
              disabled={submittingScore}
            />
          </label>
          <div className="configuration-actions">

              disabled={submittingScore}
            >
              {submittingScore ? "Submitting..." : "Submit expected score"}
            </button>
          </div>
        </div>
        {error && <p className="error">{error}</p>}

  if (session.state === "completed") {
    return (

function ScoreChart({
  history,
}: {
  history: HistoryItem[];
}): ReactElement | null {
  const scored = history
    .filter(
      (item): item is HistoryItem & { actualScore: number } =>
        item.actualScore != null,
    )
    .slice(0, 8)
    .reverse();
  if (scored.length < 2) return null;

  const width = 640;
  const height = 120;
  const padding = 12;
  const step = (width - padding * 2) / (scored.length - 1);
  const points = scored.map((item, index) => {
    const x = padding + step * index;
    const y =
      height - padding - (item.actualScore / 100) * (height - padding * 2);
    return { x, y, score: item.actualScore };
  });
  const path = points
    .map((point, index) => `${index === 0 ? "M" : "L"}${point.x},${point.y}`)
    .join(" ");

  return (
    <svg
      className="progress-chart"
      viewBox={`0 0 ${width} ${height}`}
      width="100%"
      height={height}
      role="img"
      aria-label={`Actual score across your last ${scored.length} completed interviews`}
    >
      <path d={path} fill="none" stroke="var(--accent)" strokeWidth="2" />
      {points.map((point) => (
        <circle
          key={point.x}
          cx={point.x}
          cy={point.y}
          r="3.2"
          fill="var(--accent-strong)"
        />
      ))}
    </svg>
  );
}

export type InterviewConfiguration = {

  duration: string;
  language: string;

export const initialConfiguration: InterviewConfiguration = {

  difficulty: "",
  duration: "",
  language: "Hinglish",
  mode: "Text",
};

export function validateConfiguration(
  configuration: InterviewConfiguration,
): Partial<Record<keyof InterviewConfiguration, string>> {
  const errors: Partial<Record<keyof InterviewConfiguration, string>> = {};

  if (
    !interviewTypes.includes(
      configuration.interviewType as (typeof interviewTypes)[number],
    )
  ) {
    errors.interviewType = "Choose an interview type.";
  }
  if (!configuration.topic.trim()) {
    errors.topic = "Enter a topic or skill to practice.";
  }
  if (
    !difficulties.includes(
      configuration.difficulty as (typeof difficulties)[number],
    )
  ) {
    errors.difficulty = "Choose a difficulty level.";
  }
  if (
    !durations.includes(configuration.duration as (typeof durations)[number])
  ) {
    errors.duration = "Choose an interview duration.";
  }
  if (
    !languages.includes(configuration.language as (typeof languages)[number])
  ) {
    errors.language = "Choose a language.";
  }
  if (!modes.includes(configuration.mode as (typeof modes)[number])) {
    errors.mode = "Choose an interview mode.";
  }

  return errors;
}

type InterviewConfigurationProps = {
  configuration: InterviewConfiguration;
  onChange: (configuration: InterviewConfiguration) => void;
  onAction: (
    action: ConfigurationAction,
    configuration?: InterviewConfiguration,
  ) => void;
};

  >({});

    field: keyof InterviewConfiguration,
    event: ChangeEvent<HTMLInputElement | HTMLSelectElement>,

    onChange({ ...configuration, [field]: event.target.value });

// Single point of contact with the Fastify backend. Every service module
// goes through this — never call fetch() directly from a page or hook.
//
// Auth model assumption: the backend issues an httpOnly session cookie on
// login (not a token the frontend stores/attaches). `credentials: "include"`
// is set on every request so that cookie always rides along. If the actual
// backend instead returns a bearer token, swap that in here — it's the only
// file that needs to change.

const API_BASE = import.meta.env.VITE_API_BASE_URL ?? "";

export class ApiError extends Error {
  status: number;
  body: unknown;

  constructor(status: number, message: string, body?: unknown) {
    super(message);
    this.name = "ApiError";
    this.status = status;
    this.body = body;
  }
}

interface RequestOptions {
  method?: "GET" | "POST" | "PUT" | "PATCH" | "DELETE";
  body?: unknown;
  signal?: AbortSignal;
}

export async function apiRequest<TResponse>(
  path: string,
  options: RequestOptions = {},
): Promise<TResponse> {
  const { method = "GET", body, signal } = options;
  const requestBody = body ?? (method === "POST" ? {} : undefined);

  const response = await fetch(`${API_BASE}${path}`, {
    method,
    credentials: "include",
    headers: requestBody ? { "Content-Type": "application/json" } : undefined,
    body: requestBody ? JSON.stringify(requestBody) : undefined,
    signal,
  });

  const contentType = response.headers.get("content-type") ?? "";
  const isJson = contentType.includes("application/json");
  const payload = isJson
    ? await response.json().catch(() => null)
    : await response.text().catch(() => null);

  if (!response.ok) {
    // The backend is the source of truth for error shape/messages. Adjust
    // this extraction once the real error envelope is confirmed.
    const message =
      (isJson && payload && typeof payload === "object" && "error" in payload
        ? String((payload as { error: unknown }).error)
        : isJson &&
            payload &&
            typeof payload === "object" &&
            "message" in payload
          ? String((payload as { message: unknown }).message)
          : null) ?? `Request failed with status ${response.status}`;
    throw new ApiError(response.status, message, payload);
  }

  return payload as TResponse;
}

export interface MicrophoneCapture {
  start(): Promise<void>;
  stop(): void;
  isActive(): boolean;
  onAudio(handler: (audio: ArrayBuffer) => void): void;
}

export function createMicrophoneCapture(): MicrophoneCapture {
  let stream: MediaStream | null = null;
  let audioContext: AudioContext | null = null;
  let source: MediaStreamAudioSourceNode | null = null;
  let processor: ScriptProcessorNode | null = null;
  let active = false;
  let audioHandler: ((audio: ArrayBuffer) => void) | null = null;

  return {
    async start() {
      if (active) return;

      stream = await navigator.mediaDevices.getUserMedia({
        audio: {
          echoCancellation: true,
          noiseSuppression: true,
          autoGainControl: true,
        },
      });

      audioContext = new AudioContext({ sampleRate: 16000 });
      source = audioContext.createMediaStreamSource(stream);
      processor = audioContext.createScriptProcessor(2048, 1, 1);

      processor.onaudioprocess = (event) => {
        if (!active || !audioHandler) return;

        const input = event.inputBuffer.getChannelData(0);
        const pcm = new Int16Array(input.length);

        for (let i = 0; i < input.length; i += 1) {
          const sample = Math.max(-1, Math.min(1, input[i]));
          pcm[i] = sample < 0 ? sample * 0x8000 : sample * 0x7fff;
        }

        audioHandler(pcm.buffer);
      };

      source.connect(processor);
      processor.connect(audioContext.destination);
      active = true;
    },

    stop() {
      active = false;

      processor?.disconnect();
      source?.disconnect();
      stream?.getTracks().forEach((track) => track.stop());

      void audioContext?.close();

      processor = null;
      source = null;
      stream = null;
      audioContext = null;
    },

    isActive() {
      return active;
    },

    onAudio(handler) {
      audioHandler = handler;
    },
  };
}

import type {
  VoiceErrorCode,
  VoiceSession,
  VoiceState,
  VoiceStateSnapshot,
} from "@/types/voice";
import { createMicrophoneCapture } from "./microphone";

export interface VoiceSessionOptions {
  url: string;
  sessionId: string;
  onStateChange?: (snapshot: VoiceStateSnapshot) => void;
  onTranscript?: (transcript: string, isFinal: boolean) => void;
  onError?: (code: VoiceErrorCode, message: string) => void;
}

export function createVoiceSession(
  options: VoiceSessionOptions,
): VoiceSession {
  const microphone = createMicrophoneCapture();

  let socket: WebSocket | null = null;
  let state: VoiceState = "idle";
  let errorCode: VoiceErrorCode | undefined;
  let errorMessage: string | undefined;

  const updateState = (
    nextState: VoiceState,
    nextErrorCode?: VoiceErrorCode,
    nextErrorMessage?: string,
  ) => {
    state = nextState;
    errorCode = nextErrorCode;
    errorMessage = nextErrorMessage;

    options.onStateChange?.({
      state,
      ...(errorCode ? { errorCode } : {}),
      ...(errorMessage ? { errorMessage } : {}),
    });
  };

  microphone.onAudio((audio) => {
    if (socket?.readyState === WebSocket.OPEN) {
      socket.send(audio);
    }
  });

  return {
    async connect() {
      if (socket?.readyState === WebSocket.OPEN) return;

      updateState("connecting");

      await new Promise<void>((resolve, reject) => {
        const separator = options.url.includes("?") ? "&" : "?";
        socket = new WebSocket(`${options.url}${separator}sessionId=${encodeURIComponent(options.sessionId)}`);

        socket.onopen = () => {
          updateState("listening");
          resolve();
        };

        socket.onerror = () => {
          updateState(
            "error",
            "CONNECTION_FAILED",
            "Voice connection failed.",
          );
          options.onError?.(
            "CONNECTION_FAILED",
            "Voice connection failed.",
          );
          reject(new Error("Voice connection failed"));
        };

        socket.onclose = () => {
          if (state !== "error") {
            updateState("disconnected");
          }
        };

        socket.onmessage = (event) => {
          try {
            const message = JSON.parse(event.data);

            if (message.type === "state") {
              updateState(message.state);
            } else if (message.type === "transcription") {
              options.onTranscript?.(
                message.transcript,
                Boolean(message.isFinal),
              );
            } else if (message.type === "error") {
              updateState(
                "error",
                message.code,
                message.message,
              );
              options.onError?.(message.code, message.message);
            } else if (message.type === "interrupted") {
              updateState("interrupted");
            }
          } catch {
            updateState(
              "error",
              "CONNECTION_FAILED",
              "Invalid voice message.",
            );
          }
        };
      });
    },

    async disconnect() {
      microphone.stop();
      socket?.close();
      socket = null;
      updateState("disconnected");
    },

    async startListening() {
      if (!socket || socket.readyState !== WebSocket.OPEN) {
        throw new Error("Voice session is not connected");
      }

      await microphone.start();
      updateState("listening");
    },

    async stopListening() {
      microphone.stop();

      if (socket?.readyState === WebSocket.OPEN) {
        socket.send(JSON.stringify({ type: "stopListening" }));
      }

      updateState("processing");
    },

    async interrupt() {
      microphone.stop();

      if (socket?.readyState === WebSocket.OPEN) {
        socket.send(JSON.stringify({ type: "interrupt" }));
      }

      updateState("interrupted");
    },

    getState() {
      return {
        state,
        ...(errorCode ? { errorCode } : {}),
        ...(errorMessage ? { errorMessage } : {}),
      };
    },
  };
}
